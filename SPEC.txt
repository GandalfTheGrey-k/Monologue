Monologue (Version .1) — Public RP & RP-PvP Engine
Modular Engine & Implementation Blueprint for World of Warcraft (Modern 12.0+ API)


This document presents the complete architectural specification and source implementation files for Monologue (Version .1). Monologue is a lightweight, zero-taint, modular roleplay engine for World of Warcraft Forever designed to execute real-time outgoing chat dialect translation and cooldown-gated reactive character speech.
1. Executive Summary & Design Constraints
* Target API: Modern World of Warcraft client (12.0+ native API execution). Zero legacy or Classic 1.12/TBC compatibility fallbacks. Built specifically for WoW Forever's roster of 9 playable races (Human, Dwarf, NightElf, Gnome, Orc, Scourge, Tauren, Troll, Skyborne) and 9 playable classes (WARRIOR, PALADIN, HUNTER, ROGUE, PRIEST, SHAMAN, MAGE, WARLOCK, DRUID).
* Version Pinning: Strictly maintained at Monologue_v.1. Version incrementing is disallowed to maintain addon signature stability across updates.
* Decoupled Architecture: Strict separation between engine execution logic and character data tables. Core code contains zero hardcoded race, class, or player profiles. All data files are pure, side-effect-free Lua data schemas.
* Safety & Taint Mitigation: Centralized speech routing through an isolated "Voice Box" enforcing global cooldowns, stealth protection (IsStealthed()), deck-shuffle anti-repeat mechanics, and automatic encounter safe-mode downgrades (ADDON_RESTRICTION_STATE_CHANGED).
* WoW Forever Lore & Spell Timeline Guardrail: Strictly restrict all dialogue, dialect flavor, historical references, and spell triggers to WoW Forever canon (Classic Azeroth plus WoW Forever's custom additions such as Zephras Isle, the Skyborne [High Order & Windshapers], Undead Paladins, and Dwarf Shamans). Characters MUST NOT reference future retail expansions (e.g., Outland/The Burning Crusade, Northrend/The Lich King's fall, Cataclysm, Pandaria, Shadowlands, Dragon Isles) or trigger on spells/abilities that do not exist in WoW Forever.

MonologueEngine = {

-- Deep Merge Utility for 4-Layer Profile Composition
local function DeepMerge(target, source)
    if type(target) ~= "table" then target = {} end
    if type(source) ~= "table" then return target end


    for k, v in pairs(source) do
        if type(v) == "table" then
            target[k] = DeepMerge(target[k] or {}, v)
        else
            target[k] = v
        end
    end
    return target
end


-- Bloat Prevention & Long-Term Ledger Hygiene Engine
function MonologueEngine:PruneRivals(maxEntries)
    if not MonologueDB.Rivals then return end


    maxEntries = maxEntries or 100
    local now = GetServerTime()
    local dayInSeconds = 86400


    -- Pass 1: Evict stale single-clash and non-Nemesis entries
    for name, rival in pairs(MonologueDB.Rivals) do
        local totalClashes = (rival.kills or 0) + (rival.deaths or 0)
        local ageInDays = (now - (rival.lastSeen or 0)) / dayInSeconds


        if totalClashes == 1 and ageInDays > 30 then
            MonologueDB.Rivals[name] = nil
        elseif totalClashes < 5 and ageInDays > 90 then
            MonologueDB.Rivals[name] = nil
        end
    end


    -- Pass 2: Enforce strict LRU cap protecting Nemesis and pinned rivals
    local rivalList = {}
    for name, rival in pairs(MonologueDB.Rivals) do
        table.insert(rivalList, { name = name, rival = rival })
    end


    if #rivalList > maxEntries then
        table.sort(rivalList, function(a, b)
            local aIsNemesis = ((a.rival.kills or 0) + (a.rival.deaths or 0)) >= 5 or a.rival.pinned
            local bIsNemesis = ((b.rival.kills or 0) + (b.rival.deaths or 0)) >= 5 or b.rival.pinned
            if aIsNemesis ~= bIsNemesis then
                return aIsNemesis
            end
            return (a.rival.lastSeen or 0) > (b.rival.lastSeen or 0)
        end)


        for i = maxEntries + 1, #rivalList do
            MonologueDB.Rivals[rivalList[i].name] = nil
        end
    end
end


-- Deck-Shuffle Pool Retrieval (Prevents raw math.random repetition)
function MonologueEngine:GetShuffledLine(categoryKey, linesTable)
    if not linesTable or #linesTable == 0 then return nil end


    if not self.ShuffledPools[categoryKey] or #self.ShuffledPools[categoryKey] == 0 then
        self.ShuffledPools[categoryKey] = {}
        for i, line in ipairs(linesTable) do
            table.insert(self.ShuffledPools[categoryKey], line)
        end
        -- Fisher-Yates Shuffle
        for i = #self.ShuffledPools[categoryKey], 2, -1 do
            local j = math.random(i)
            self.ShuffledPools[categoryKey][i], self.ShuffledPools[categoryKey][j] = 
                self.ShuffledPools[categoryKey][j], self.ShuffledPools[categoryKey][i]
        end
    end


    return table.remove(self.ShuffledPools[categoryKey], 1)
end


-- Boss Feat Unlock Helper
function MonologueEngine:UnlockFeat(featKey)
    if not featKey then return end
    MonologueDB.Feats = MonologueDB.Feats or {}
    MonologueDB.Feats[featKey] = true
    self:BuildActiveProfile()
    DEFAULT_CHAT_FRAME:AddMessage("|cff00ccffMonologue Feat Unlocked:|r " .. featKey)
end


-- Profile Assembly Engine
function MonologueEngine:BuildActiveProfile()
    local _, raceFile = UnitRace("player")
    local _, classFile = UnitClass("player")
    local playerName = UnitName("player")
    local level = UnitLevel("player") or 1


    local merged = {}


    -- Layer 1: Base Race
    if self.Data.Races[raceFile] then
        merged = DeepMerge(merged, self.Data.Races[raceFile])
    end


    -- Layer 2: Base Class
    if self.Data.Classes[classFile] then
        merged = DeepMerge(merged, self.Data.Classes[classFile])
    end


    -- Layer 3: Dynamic Growth Arc & Persona Overrides
    local arcStage = "stage1"
    local stageKey = "Stage1_Early"
    if level >= 50 then
        arcStage = "stage3"
        stageKey = "Stage3_Late"
    elseif level >= 20 then
        arcStage = "stage2"
        stageKey = "Stage2_Mid"
    end


    local selectedPersona = MonologueDB.GrowthArc and MonologueDB.GrowthArc[arcStage] or MonologueDB.selectedPersona
    if selectedPersona and self.Data.Personas[selectedPersona] then
        merged = DeepMerge(merged, self.Data.Personas[selectedPersona])
    end


    -- Layer 4: Specific Character Profile, GrowthStages & Unlocked BossFeats
    if self.Data.Profiles[playerName] then
        local charProfile = self.Data.Profiles[playerName]
        merged = DeepMerge(merged, charProfile)


        if charProfile.GrowthStages and charProfile.GrowthStages[stageKey] then
            merged = DeepMerge(merged, charProfile.GrowthStages[stageKey])
        end


        if charProfile.BossFeats and MonologueDB.Feats then
            for featKey, featData in pairs(charProfile.BossFeats) do
                if MonologueDB.Feats[featKey] then
                    merged = DeepMerge(merged, featData)
                end
            end
        end
    end


    self.ActiveProfile = merged
end


-- Central Voice Box Router
function MonologueEngine:Speak(text, chatType, targetUnit, categoryCooldown, emoteToken)
    if not MonologueDB.enabled or not text or text == "" then return end
    if IsStealthed() or UnitIsAFK("player") then return end


    local now = GetTime()
    if now < self.Cooldowns.GlobalBreath then return end


    if categoryCooldown then
        local catKey = categoryCooldown.key or "default"
        if self.Cooldowns.Categories[catKey] and now < self.Cooldowns.Categories[catKey] then
            return
        end
        self.Cooldowns.Categories[catKey] = now + (categoryCooldown.duration or 30)
    end


    -- Enforce Global Breath Lock (45 Seconds)
    self.Cooldowns.GlobalBreath = now + 45
    self.State.LastSpeechTime = now


    -- Process Dynamic RP Tokens
    if MonologueRPBridge then
        text = MonologueRPBridge:ProcessTokens(text, targetUnit)
    end


    -- Dialect Translation Pass
    if MonologueTranslator and chatType ~= "EMOTE" then
        text = MonologueTranslator:Translate(text, MonologueDB.translationIntensity)
    end


    -- Encounter Restriction Guardrail
    if self.State.InRestrictedEncounter then
        -- Downgrade to local self-print during restricted encounter states to prevent taint
        DEFAULT_CHAT_FRAME:AddMessage("|cff00ccff[Monologue Safe-Mode]:|r " .. text)
        return
    end


    -- Outgoing Routing
    if chatType == "SAY" or chatType == "YELL" or chatType == "PARTY" then
        C_ChatInfo.SendChatMessage(text, chatType)
    elseif chatType == "EMOTE" then
        C_ChatInfo.SendChatMessage(text, "EMOTE")
    end


    -- Cross-Faction Native Emote Pairing
    if targetUnit and UnitIsEnemy("player", targetUnit) and emoteToken then
        DoEmote(emoteToken, targetUnit)
    end
end


-- Trigger Processing Engine
function MonologueEngine:ProcessTrigger(tier, categoryKey, pool, chance, targetUnit, emoteToken)
    if not pool or #pool == 0 then return end


    local effectiveChance = (chance or 1.0) * MonologueDB.chattiness
    if math.random() > effectiveChance then return end


    local line = self:GetShuffledLine(categoryKey, pool)
    if not line then return end


    local cooldownDuration = 60
    if tier == 1 then cooldownDuration = 60
    elseif tier == 2 then cooldownDuration = 90
    elseif tier == 3 then cooldownDuration = 60
    elseif tier == 4 then cooldownDuration = 180 end


    self:Speak(line, "SAY", targetUnit, { key = categoryKey, duration = cooldownDuration }, emoteToken)
end


-- Nemesis Ledger Engine
function MonologueEngine:RecordPvPInteraction(targetName, isKill)
    if not targetName then return end


    MonologueDB.Rivals = MonologueDB.Rivals or {}
    local rival = MonologueDB.Rivals[targetName] or { kills = 0, deaths = 0, lastSeen = 0, lastZone = "the wild" }


    if isKill then
        rival.kills = rival.kills + 1
    else
        rival.deaths = rival.deaths + 1
    end
    rival.lastSeen = GetServerTime()
    rival.lastZone = GetRealZoneText() or "the wild"


    MonologueDB.Rivals[targetName] = rival
    self:PruneRivals(100)
end


function MonologueEngine:GetRivalTier(targetName)
    local rival = MonologueDB.Rivals and MonologueDB.Rivals[targetName]
    local currentZone = GetRealZoneText()
    if MonologueDB.SkirmishHistory and MonologueDB.SkirmishHistory[currentZone] then return "DejaVu" end
    if not rival then return "FirstClash" end


    local total = rival.kills + rival.deaths
    if total >= 5 then return "Nemesis"
    elseif rival.kills >= 3 then return "Dominance"
    elseif rival.deaths > rival.kills then return "Vendetta"
    elseif total > 1 then return "Rematch"
    else return "FirstClash" end
end

4.3 Live Chat Dialect Translator (Monologue_Translator.lua)
--[[
    Monologue Dialect Translator (Version .1)
    Hooks outgoing chat submissions to apply phonetic dialect transformations.
    Enforces OOC protection, emote protection, and channel stepdown rules.
--]]


MonologueTranslator = {
    DefaultSwaps = {},
    DefaultPhonetics = {},
}


-- Core Translation Engine
function MonologueTranslator:Translate(text, intensity)
    if not text or text == "" or intensity == 0 then return text end
    
    -- Command & Slash Bypass
    if text:sub(1, 1) == "/" then return text end
    
    -- Preserve Emotes (*text*) and OOC ((text)) via Marker Tokens
    local preservedBlocks = {}
    local function Preserve(match)
        table.insert(preservedBlocks, match)
        return "___PRESERVED_" .. #preservedBlocks .. "___"
    end
    
    -- Mask OOC
    text = text:gsub("%(%b()%)", Preserve)
    -- Mask Asterisk Emotes
    text = text:gsub("%*.-%*", Preserve)
    
    local dialect = MonologueEngine.ActiveProfile.Dialect or {}
    
    -- Tier 1: Light Intensity (Vocabulary & Greetings Swap Only)
    if intensity >= 1 and dialect.wordSwaps then
        for word, replacement in pairs(dialect.wordSwaps) do
            local pattern = "%f[%a]" .. word .. "%f[%A]"
            text = text:gsub(pattern, replacement)
        end
    end
    
    -- Tier 2: Standard Intensity (Phonetic Rules)
    if intensity >= 2 and dialect.phonetics then
        for pattern, replacement in pairs(dialect.phonetics) do
            text = text:gsub(pattern, replacement)
        end
    end
    
    -- Tier 3: Unhinged Intensity (Suffix Additions & Fluff)
    if intensity >= 3 then
        if dialect.suffixes and #dialect.suffixes > 0 and math.random() < 0.35 then
            local suffix = dialect.suffixes[math.random(#dialect.suffixes)]
            text = text .. " " .. suffix
        end
    end
    
    -- Restore Preserved Blocks
    for i, original in ipairs(preservedBlocks) do
        text = text:gsub("___PRESERVED_" .. i .. "___", original)
    end
    
    -- Enforce Client Safe 255-Character Limit
    if #text > 255 then
        text = text:sub(1, 252) .. "..."
    end
    
    return text
end


-- Hook Modern Blizzard Chat Editbox Path
local function HookChatEditBoxes()
    for i = 1, NUM_CHAT_WINDOWS do
        local editBox = _G["ChatFrame" .. i .. "EditBox"]
        if editBox and not editBox.MonologueHooked then
            editBox:HookScript("OnEnterPressed", function(self)
                local text = self:GetText()
                local chatType = self:GetAttribute("chatType")
                
                -- Channel Auto-Stepdown Logic
                if MonologueDB.channelStepdown then
                    if chatType == "RAID" or chatType == "INSTANCE_CHAT" or chatType == "WHISPER" or chatType == "GUILD" then
                        return -- Completely bypass translation in competitive/private channels
                    elseif chatType == "PARTY" then
                        text = MonologueTranslator:Translate(text, 1) -- Step down to Light (1)
                        self:SetText(text)
                        return
                    end
                end
                
                if chatType == "SAY" or chatType == "YELL" then
                    text = MonologueTranslator:Translate(text, MonologueDB.translationIntensity)
                    self:SetText(text)
                end
            end)
            editBox.MonologueHooked = true
        end
    end
end


local hookFrame = CreateFrame("Frame")

    HookChatEditBoxes()
end)

--[[
    Monologue RP Bridge (Version .1)
    Provides soft-dependency resolution for TRP3_API and MSP (Mary Sue Protocol).
    Evaluates IC/OOC player flags and parses dynamic replacement tokens.
--]]


MonologueRPBridge = {}


-- Safely Checks IC Status via TRP3 or MSP
function MonologueRPBridge:IsPlayerIC()
    if TRP3_API and TRP3_API.profile then
        local status = TRP3_API.profile.getPlayerCurrentProfile().player.character.v or 1
        return status == 1 -- 1 = IC, 2 = OOC
    elseif msp and msp.mystatus then
        return msp.mystatus.field["IC"] == "1"
    end
    return true -- Default to IC if no external RP addon is active
end


-- Dynamic Token Resolution Engine
function MonologueRPBridge:ProcessTokens(text, targetUnit)
    if not text or text == "" then return "" end


    -- Dynamic Replacement: {rpname}
    if text:find("{rpname}") then
        local rpName = UnitName("player")
        if TRP3_API and TRP3_API.r then
            rpName = TRP3_API.r.name("player") or rpName
        end
        text = text:gsub("{rpname}", rpName)
    end


    -- Dynamic Replacement: {target}
    if text:find("{target}") then
        local tName = targetUnit and UnitName(targetUnit) or "stranger"
        if targetUnit and TRP3_API and TRP3_API.r then
            tName = TRP3_API.r.name(targetUnit) or tName
        end
        text = text:gsub("{target}", tName)
    end


    -- Dynamic Replacement: {zone}
    if text:find("{zone}") then
        local zoneName = GetRealZoneText() or "the wild"
        text = text:gsub("{zone}", zoneName)
    end


    -- Dynamic Replacement: {lastzone}
    if text:find("{lastzone}") then
        local lastZoneName = "our last clash"
        if targetUnit then
            local tName = UnitName(targetUnit)
            if tName and MonologueDB and MonologueDB.Rivals and MonologueDB.Rivals[tName] then
                lastZoneName = MonologueDB.Rivals[tName].lastZone or lastZoneName
            end
        end
        text = text:gsub("{lastzone}", lastZoneName)
    end


    -- Dynamic Replacement: {rank}
    if text:find("{rank}") then
        local rankName = "Member"
        if IsInGuild() then
            rankName = select(1, GetGuildInfo("player")) or rankName
        end
        text = text:gsub("{rank}", rankName)
    end


    -- Dynamic Replacement: {opener}
    if text:find("{opener}") then
        local fragments = MonologueEngine.ActiveProfile.Fragments
        local openerPool = fragments and fragments.openers
        local opener = (openerPool and MonologueEngine:GetShuffledLine("Fragment_Opener", openerPool)) or ""
        text = text:gsub("{opener}", opener)
    end


    -- Dynamic Replacement: {closer}
    if text:find("{closer}") then
        local fragments = MonologueEngine.ActiveProfile.Fragments
        local closerPool = fragments and fragments.closers
        local closer = (closerPool and MonologueEngine:GetShuffledLine("Fragment_Closer", closerPool)) or ""
        text = text:gsub("{closer}", closer)
    end


    return text
end
